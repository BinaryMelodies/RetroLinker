186 void Repeat(uint32_t block_count, uint32_t repeat_count)
187 {
188 // unless this is the last iteration of this block
189 if(current_repeat_count != 1)
190 {
191 // jump over this (already processed) instruction and the block
192 reloc_instr_ptr --;
193 Regress(block_count);
194 }
195
196 if(current_repeat_count == 0)
197 {
198 // iteration starts
199 current_repeat_count = repeat_count;
